  <p>Tumble Boy es una actividad libre del ecosistema <b>Sugar Labs/OLPC</b>, nacida en un game jam para la XO en 2007 y popularizada en Uruguay con las laptops XO del Plan Ceibal. Esta versión <b>Reborn</b> lo rescata y lo amplía: <b>21 niveles originales</b> + editor de niveles visual + packs descargables, con tres zócalos de guardado para no perder nada.</p>

  <p>Corre nativo en <b>Android</b> (teléfonos, tablets y <b>Android TV</b>) y en escritorio Linux, y está hecho con <b>Godot 3.6 + GLES2</b> para ser ligero incluso en hardware humilde.</p>

  <!-- CAPTURA 2: Editor de niveles con bloques pintados -->
  <img src="https://cdn.jsdelivr.net/gh/Aleperix/tumbleboy-reborn@main/blog/xo-galaxy/editor.png" alt="Editor de niveles visual de TumbleBoy Reborn">
  <p class="tbr-caption">El editor visual: pinta bloques, meta y jugador, y guarda tu nivel.</p>

  <h2>Características</h2>
  <ul>
    <li><b>Botón Atrás de Android</b> navega hacia atrás: en el menú principal cierra la app, en submenús, partidas y editor vuelve a la pantalla anterior.</li>
    <li><b>Controles táctiles</b>: joystick virtual y switch Mando ↔ Acelerómetro, recordados entre partidas.</li>
    <li><b>Android TV</b>: foco completo con D-pad y botones, listo para el sillón.</li>
    <li><b>Packs comunitarios</b>: descarga niveles hechos por la comunidad (<code>numeros</code>, <code>letras</code>, …).</li>
    <li><b>Modo historia + partida rápida</b> y <b>3 zócalos de guardado</b>.</li>
  </ul>

  <h2>Instalación paso a paso</h2>
  <ol>
    <li>Descarga el APK de la <b>ARM64</b> (o el que corresponda a tu dispositivo).</li>
    <li>En el teléfono, abre Ajustes → Seguridad y activa <b>&laquo;Orígenes desconocidos&raquo;</b> (o &laquo;Permitir esta fuente&raquo;) para el navegador o el gestor de archivos.</li>
    <li>Toca el APK descargado y pulsa <b>Instalar</b>.</li>
    <li>Para <b>Android TV</b>: copia el APK a un USB y usa un gestor de archivos, o envíalo por ADB (<code>adb install tumbleboy-reborn-ARM64.apk</code>).</li>
    <li>Si ya tenías una versión anterior, la instalación <b>actualiza</b> sin borrar tus guardados.</li>
  </ol>

  (function () {
    var REPO = "Aleperix/tumbleboy-reborn";
    var KEY = "tbr_releases";
    var CACHE_SCHEMA = 2;
    var TTL = 12 * 3600e3;

    var $ = function (id) { return document.getElementById(id); };
    var dot = $("tbrDot"), status = $("tbrStatus"), note = $("tbrChangelogNote"),
        acc = $("tbrChangelog"), refreshBtn = $("tbrRefresh");

    function readCache() {
      try {
        var raw = localStorage.getItem(KEY);
        if (!raw) return null;
        var rec = JSON.parse(raw);
        if (!rec || rec.schema !== CACHE_SCHEMA || !rec.t || Date.now() - rec.t > TTL) return null;
        return rec;
      } catch (e) { return null; }
    }

    function writeCache(data, etag) {
      try {
        var rec = { t: Date.now(), schema: CACHE_SCHEMA, data: data, etag: etag || "" };
        localStorage.setItem(KEY, JSON.stringify(rec));
      } catch (e) {}
    }

    function esc(s) {
      return String(s).replace(/[&<>"']/g, function (c) {
        return { "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c];
      });
    }

    function inline(s) {
      return esc(s)
        .replace(/\*\*([^*]+)\*\*/g, "<b>$1</b>")
        .replace(/`([^`]+)`/g, "<code>$1</code>")
        .replace(/\[([^\]]+)\]\((https?:\/\/[^)\s]+)\)/g, '<a href="$2">$1</a>');
    }

    function mdToHtml(md) {
      if (!md) return "";
      var out = [], para = [];
      var flush = function () { if (para.length) { out.push("<p>" + para.join(" ") + "</p>"); para = []; } };
      md.split(/\r?\n/).forEach(function (line) {
        var t = line.trim();
        if (!t) { flush(); return; }
        var h = /^(#{1,4})\s+(.*)$/.exec(t);
        if (h) { flush(); out.push("<h4>" + inline(h[2]) + "</h4>"); return; }
        if (t.indexOf("- ") === 0 || t.indexOf("* ") === 0) {
          flush();
          var lis = [];
          md.split(/\r?\n/).forEach(function (l2) {
            var tt = l2.trim();
            if (tt.indexOf("- ") === 0 || tt.indexOf("* ") === 0) lis.push("<li>" + inline(tt.slice(2)) + "</li>");
            else if (tt) { /* ignore stray */ }
          });
          out.push("<ul>" + lis.join("") + "</ul>");
          return;
        }
        if (/^#{4,}/.test(t)) return;
        para.push(inline(t));
      });
      flush();
      return out.join("");
    }

    function fmtDate(iso) {
      if (!iso) return "";
      try {
        var d = new Date(iso);
        return d.toLocaleDateString(undefined, { year: "numeric", month: "short", day: "numeric" });
      } catch (e) { return ""; }
    }

    function updateButtons(releases) {
      var links = document.querySelectorAll("#tbrDownloads .tbr-btn");
      links.forEach(function (a) {
        var arch = a.getAttribute("data-arch");
        var name = "tumbleboy-reborn-" + arch + ".apk";
        var size = a.querySelector(".tbr-btn-size");
        var parts = [];
        if (releases && releases.length) {
          var r = releases[0];
          if (r.tag_name) parts.push(r.tag_name);
          var asset = (r.assets || []).filter(function (x) { return x.name === name; })[0];
          if (asset && asset.size) parts.push(Math.round(asset.size / 1048576) + " MB");
        }
        if (size) size.textContent = parts.join("\u00b7");
      });
    }

    function setStatus(releases, rate) {
      var latest = releases && releases[0];
      if (latest && latest.tag_name) {
        dot.className = "tbr-dot on";
        status.textContent = "Publicado en GitHub: " + latest.tag_name +
          (releases.length > 1 ? "\u00b7 " + releases.length + " versiones" : "") +
          (rate ? "\u00b7 " + rate + " consultas restantes" : "");
      } else {
        dot.className = "tbr-dot off";
        status.textContent = "Sin conexi\u00f3n: los botones llevan a la versi\u00f3n publicada en GitHub.";
      }
    }

    function render(releases, rate) {
      note.style.display = "none";
      acc.innerHTML = "";
      releases = releases.slice(0, 12);
      releases.forEach(function (r, i) {
        var body = mdToHtml(r.body || "");
        var assets = (r.assets || []).map(function (a) {
          return '<div><a href="' + esc(a.browser_download_url) + '">' + esc(a.name) +
                 '</a> <span class="tbr-muted">(' + Math.round((a.size || 0) / 1048576) + " MB)</span></div>";
        }).join("");
        var d = document.createElement("details");
        if (i === 0) d.setAttribute("open", "");
        d.innerHTML = '<summary><span>' + esc(r.tag_name || r.name) +
          '</span><span class="tbr-acc-date">' + fmtDate(r.published_at) + "</span></summary>" +
          '<div class="tbr-acc-body">' + body + (assets ? "<div>" + assets + "</div>" : "") + "</div>";
        acc.appendChild(d);
      });
      updateButtons(releases);
      setStatus(releases, rate);
    }

    var JSON_URL = "https://cdn.jsdelivr.net/gh/" + REPO + "@main/blog/xo-galaxy/releases.json";
    var API_URL = "https://api.github.com/repos/" + REPO + "/releases?per_page=100";

    function fetchReleases(force) {
      var cached = readCache();
      if (cached && !force) {
        render(cached.data);
        return;
      }
      setStatus(null);
      fetch(JSON_URL, { cache: "no-store" })
        .then(function (res) {
          if (!res.ok) throw new Error("jsdelivr " + res.status);
          return res.json();
        })
        .then(function (data) {
          if (!Array.isArray(data) || !data.length) throw new Error("no data");
          writeCache(data, "");
          render(data);
        })
        .catch(function () {
          fetchGithub(cached);
        });
    }

    function fetchGithub(cached) {
      var headers = {};
      if (cached && cached.etag) headers["If-None-Match"] = cached.etag;
      fetch(API_URL, { headers: headers })
        .then(function (res) {
          if (res.status === 304 && cached) {
            writeCache(cached.data, cached.etag);
            render(cached.data);
            return;
          }
          if (!res.ok) {
            if (cached) render(cached.data);
            else setStatus(null);
            return;
          }
          var etag = res.headers.get("ETag") || "";
          var remaining = res.headers.get("X-RateLimit-Remaining");
          return res.json().then(function (data) {
            if (!Array.isArray(data) || !data.length) throw new Error("no data");
            writeCache(data, etag);
            render(data, remaining);
          });
        })
        .catch(function () {
          if (cached) render(cached.data);
          else setStatus(null);
        });
    }

    refreshBtn.addEventListener("click", function () { fetchReleases(true); });
    var c = readCache();
    if (c) render(c.data);
    fetchReleases(false);
  })();
  </script>
